cmake_minimum_required(VERSION 3.20)
project(td3color VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(TD3_SANITIZE "Enable address and undefined behavior sanitizers" OFF)
add_library(td3 src/graph.cpp src/decomposition.cpp src/elimination.cpp src/solver.cpp src/report.cpp)
target_include_directories(td3 PUBLIC include)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS assets/report.html)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/assets/report.html" TD3_REPORT_HTML)
# MSVC limits the size of each string literal. Adjacent raw literals preserve
# the exact template while keeping each source line in a separate literal.
string(REPLACE "\n" "\n)_TD3_REPORT_\"\nR\"_TD3_REPORT_(" TD3_REPORT_HTML "${TD3_REPORT_HTML}")
configure_file(src/report_template.hpp.in generated/report_template.hpp @ONLY)
target_include_directories(td3 PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
if(MINGW)
  target_link_options(td3 PUBLIC -static)
endif()
if(MSVC)
  target_compile_options(td3 PRIVATE /W4 /permissive-)
  target_compile_options(td3 PUBLIC /utf-8)
else()
  target_compile_options(td3 PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
  if(TD3_SANITIZE)
    target_compile_options(td3 PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(td3 PUBLIC -fsanitize=address,undefined)
  endif()
endif()
add_executable(td3color src/main.cpp)
target_link_libraries(td3color PRIVATE td3)
include(CTest)
if(BUILD_TESTING)
  add_executable(td3_tests tests/tests.cpp)
  target_link_libraries(td3_tests PRIVATE td3)
  target_compile_definitions(td3_tests PRIVATE TD3_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
  add_test(NAME unit_and_fixtures COMMAND td3_tests)
  add_test(NAME exhaustive_six_vertices COMMAND td3_tests --exhaustive)
  add_test(NAME cli_integration COMMAND ${CMAKE_COMMAND} -DEXE=$<TARGET_FILE:td3color> -DSOURCE=${CMAKE_CURRENT_SOURCE_DIR} -DBINARY=${CMAKE_CURRENT_BINARY_DIR} -P ${CMAKE_CURRENT_SOURCE_DIR}/tests/cli.cmake)
  add_executable(td3_benchmark tests/benchmark.cpp)
  target_link_libraries(td3_benchmark PRIVATE td3)
  target_compile_definitions(td3_benchmark PRIVATE TD3_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
  add_test(NAME benchmark_smoke COMMAND td3_benchmark --quick)
  set_tests_properties(benchmark_smoke PROPERTIES PASS_REGULAR_EXPRESSION "table_stress")
endif()
install(TARGETS td3color RUNTIME DESTINATION bin)
